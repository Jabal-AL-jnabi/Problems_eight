#include <iostream>
using namespace std;

short readYear ()
{
    short year;
    cout<<"please enter year check?\n";
    cin>>year;
    
    return year;
}
bool isBearYears (short year)
{
    if(year%400==0)
    {
        return 1;
    }
    else if (year%100==0)
    {
        return 0;
    }
    else if (year%4==0)
    {
        return 1;
    }
    else
    {
        return 0;
    }
}
int main()
{
    short year=readYear();
    if(isBearYears(year))
    {
        cout << "\nYes, Year [" << year << "] is a leap year. \n" ;
    }
    else
    {
       cout << "\nNo, Year [" << year << "] is NOT a leap year. \n" ;
    }
    
    return 0;
}
