bool isleapYears (short year)
{
    return (year%400==0||year%100!=0&&year%4==0);
}
short dayInMonth(short year,short month)
{
   if(month==2)
   {
       return (isleapYears(year))?29:28;
   }
   short month31Days[7]={1,3,5,7,8,10,12};
   for(short i=1;i<7;++i)
   {
     if(month==month31Days[i]) 
     return 31;
   }
   return 30;
}
int huorInMonth(short year,short month)
{
    return dayInMonth(year,month)*24; 
}
int minutesInMonth(short year,short month)
{
    return huorInMonth(year,month)*60; 
}
int secondInMonth(short year,short month)
{
    return minutesInMonth(year,month)*60; 
}
