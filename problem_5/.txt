bool isleapYears (short year)
{
    return (year%400==0||year%100!=0&&year%4==0);
}
short dayInMunth(short year,short munth)
{
   if(munth==2)
   {
       return (isleapYears(year))?29:28;
   }
   else
   {
       return ((munth<=7&&munth%2==1)||(munth>7&&munth%2==0))?31:30;
   }
}
int huorInMunth(short year,short munth)
{
    return dayInMunth(year,munth)*24; 
}
int minutesInMunth(short year,short munth)
{
    return huorInMunth(year,munth)*60; 
}
int secondInMunth(short year,short munth)
{
    return minutesInMunth(year,munth)*60; 
}
